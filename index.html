Practical 3

Conditional Statements: if-else, else-if, nested if-else and switch

import java.util.Scanner;

public class Practical3 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // a) Even or Odd using if-else
        System.out.print("Enter a number: ");
        int n = sc.nextInt();
        if (n % 2 == 0)
            System.out.println("Even");
        else
            System.out.println("Odd");

        // b) Roots of a quadratic equation using else-if
        System.out.print("Enter a, b and c: ");
        double a = sc.nextDouble();
        double b = sc.nextDouble();
        double c = sc.nextDouble();

        double d = b * b - 4 * a * c;
        if (d > 0) {
            double r1 = (-b + Math.sqrt(d)) / (2 * a);
            double r2 = (-b - Math.sqrt(d)) / (2 * a);
            System.out.println("Roots: " + r1 + ", " + r2);
        } else if (d == 0) {
            double r = -b / (2 * a);
            System.out.println("Equal root: " + r);
        } else {
            System.out.println("Complex roots");
        }

        // c) Largest of three using nested if-else
        System.out.print("Enter three numbers: ");
        int p = sc.nextInt(), q = sc.nextInt(), r = sc.nextInt();
        int largest;
        if (p > q) {
            if (p > r) largest = p;
            else largest = r;
        } else {
            if (q > r) largest = q;
            else largest = r;
        }
        System.out.println("Largest: " + largest);

        // d) Week day using switch
        System.out.print("Enter day number (1-7): ");
        int day = sc.nextInt();
        switch (day) {
            case 1: System.out.println("Monday"); break;
            case 2: System.out.println("Tuesday"); break;
            case 3: System.out.println("Wednesday"); break;
            case 4: System.out.println("Thursday"); break;
            case 5: System.out.println("Friday"); break;
            case 6: System.out.println("Saturday"); break;
            case 7: System.out.println("Sunday"); break;
            default: System.out.println("Invalid day");
        }

        sc.close();
    }
}
Practical 4

Looping Constructs: while, do-while and for loop

import java.util.Scanner;

public class Practical4 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // a) Reverse a number using while loop
        System.out.print("Enter number to reverse: ");
        int n = sc.nextInt();
        int temp = n, reverse = 0;
        while (temp != 0) {
            reverse = reverse * 10 + temp % 10;
            temp /= 10;
        }
        System.out.println("Reverse: " + reverse);

        // b) Prime number using do-while loop
        System.out.print("Enter number to check prime: ");
        int num = sc.nextInt();
        int i = 2;
        boolean prime = num > 1;
        if (num > 1) {
            do {
                if (num % i == 0 && i != num) {
                    prime = false;
                    break;
                }
                i++;
            } while (i <= Math.sqrt(num));
        }
        System.out.println(prime ? "Prime" : "Not Prime");

        // c) nth Fibonacci term using for loop
        System.out.print("Enter n: ");
        int nTerm = sc.nextInt();
        long first = 0, second = 1, fib = 0;
        if (nTerm == 1) fib = 0;
        else if (nTerm == 2) fib = 1;
        else {
            for (int j = 3; j <= nTerm; j++) {
                fib = first + second;
                first = second;
                second = fib;
            }
        }
        System.out.println("Fibonacci term: " + (nTerm == 1 ? first : nTerm == 2 ? second : fib));

        sc.close();
    }
}
Practical 5

Methods and Recursion

public class Practical5 {
    // Method
    static int add(int a, int b) {
        return a + b;
    }

    // Recursive method
    static int factorial(int n) {
        if (n <= 1)
            return 1;
        return n * factorial(n - 1);
    }

    public static void main(String[] args) {
        System.out.println("Sum = " + add(10, 20));
        System.out.println("Factorial = " + factorial(5));
    }
}

Practical 7

1D and 2D Arrays: Maximum/Second Maximum, Sorting Names, Matrix Addition and 3×3 Multiplication

import java.util.Arrays;

public class Practical7 {
    public static void main(String[] args) {
        // a) Maximum and second maximum without duplicates
        int[] arr = {10, 50, 20, 50, 30, 40};
        int max = Integer.MIN_VALUE, second = Integer.MIN_VALUE;

        for (int x : arr) {
            if (x > max) {
                second = max;
                max = x;
            } else if (x > second && x != max) {
                second = x;
            }
        }
        System.out.println("Maximum: " + max);
        System.out.println("Second Maximum: " + second);

        // b) Sort names in ascending order
        String[] names = {"Rohit", "Aman", "Karan", "Vikas"};
        Arrays.sort(names);
        System.out.println("Sorted names: " + Arrays.toString(names));

        // c) Addition of two matrices
        int[][] A = {{1, 2}, {3, 4}};
        int[][] B = {{5, 6}, {7, 8}};
        int[][] sum = new int[2][2];

        System.out.println("Matrix Addition:");
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < 2; j++) {
                sum[i][j] = A[i][j] + B[i][j];
                System.out.print(sum[i][j] + " ");
            }
            System.out.println();
        }

        // d) 3x3 matrix multiplication
        int[][] X = {
            {1, 2, 3},
            {4, 5, 6},
            {7, 8, 9}
        };
        int[][] Y = {
            {9, 8, 7},
            {6, 5, 4},
            {3, 2, 1}
        };
        int[][] product = new int[3][3];

        System.out.println("3x3 Matrix Multiplication:");
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                for (int k = 0; k < 3; k++)
                    product[i][j] += X[i][k] * Y[k][j];
                System.out.print(product[i][j] + " ");
            }
            System.out.println();
        }
    }
}

Practical 12

Inheritance: Single, Multilevel, Hierarchical and Hybrid

class Animal {
    void eat() {
        System.out.println("Animal eats");
    }
}

// Single + Multilevel inheritance
class Dog extends Animal {
    void bark() {
        System.out.println("Dog barks");
    }
}

class Puppy extends Dog {
    void play() {
        System.out.println("Puppy plays");
    }
}

// Hierarchical inheritance
class Cat extends Animal {
    void meow() {
        System.out.println("Cat meows");
    }
}

// Hybrid inheritance is achieved using classes + interfaces
interface Pet {
    void friendly();
}

class PetDog extends Dog implements Pet {
    public void friendly() {
        System.out.println("Pet dog is friendly");
    }
}

public class Practical12 {
    public static void main(String[] args) {
        // Single
        Dog d = new Dog();
        d.eat();
        d.bark();

        // Multilevel
        Puppy p = new Puppy();
        p.eat();
        p.bark();
        p.play();

        // Hierarchical
        Cat c = new Cat();
        c.eat();
        c.meow();

        // Hybrid
        PetDog pd = new PetDog();
        pd.eat();
        pd.bark();
        pd.friendly();
    }
}

Practical 20

Queue Interface using PriorityQueue and Deque

import java.util.Deque;
import java.util.PriorityQueue;
import java.util.ArrayDeque;
import java.util.Queue;

public class Practical20 {
    public static void main(String[] args) {
        // PriorityQueue
        Queue<Integer> priorityQueue = new PriorityQueue<>();
        priorityQueue.add(30);
        priorityQueue.add(10);
        priorityQueue.add(20);

        System.out.println("PriorityQueue: " + priorityQueue);
        System.out.println("Removed: " + priorityQueue.poll());
        System.out.println("After removal: " + priorityQueue);

        // Deque
        Deque<String> deque = new ArrayDeque<>();
        deque.addFirst("A");
        deque.addLast("B");
        deque.addFirst("C");

        System.out.println("Deque: " + deque);
        System.out.println("First: " + deque.peekFirst());
        System.out.println("Last: " + deque.peekLast());
    }
}

    @Override
    public void run() {
        for (int i = 1; i <= 5; i++)
            System.out.println("Thread class: " + i);
    }

class MyRunnable implements Runnable {
